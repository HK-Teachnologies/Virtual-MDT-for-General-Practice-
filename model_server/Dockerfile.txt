FROM python:3.12-slim AS build

LABEL maintainer="HK Technologies"

ARG NAME=hcd-ai-model-server
ARG HOME_DIR=/usr/src/${NAME}

RUN apt-get update && apt-get install -y --no-install-recommends \
    gcc \
    git \
    python3-dev \
    && rm -rf /var/lib/apt/lists/*

RUN useradd -ms /bin/bash -d ${HOME_DIR} container_user \
    && mkdir -p ${HOME_DIR} \
    && chown -R container_user: ${HOME_DIR}

COPY requirements.txt /requirements.txt
RUN --mount=type=cache,target=/root/.cache/pip pip install torch --index-url https://download.pytorch.org/whl/cpu
RUN --mount=type=cache,target=/root/.cache/pip pip install --no-cache-dir -r /requirements.txt

COPY . ${HOME_DIR}

FROM python:3.12-slim

ARG NAME=hcd-ai-model-server
ARG PORT=8000
ARG HOME_DIR=/usr/src/${NAME}

# Models to pre-download into the image so the container starts without a
# network round-trip (or a dependency on huggingface.co being up). The compose
# file passes these as build args from the same .env values used at runtime.
ARG EMBEDDING_MODEL_NAME=Alibaba-NLP/gte-base-en-v1.5
ARG CROSS_ENCODER_MODEL=cross-encoder/ms-marco-MiniLM-L-6-v2

# Runtime defaults match the baked weights; environment variables can still
# override, but with compose passing the same values as build args they
# can't diverge.
ENV EMBEDDING_MODEL_NAME=${EMBEDDING_MODEL_NAME} \
    CROSS_ENCODER_MODEL=${CROSS_ENCODER_MODEL}

COPY --from=build /usr/local/lib/python3.12/site-packages /usr/local/lib/python3.12/site-packages
COPY --from=build /usr/local/bin /usr/local/bin
COPY --from=build ${HOME_DIR} ${HOME_DIR}

ENV PYTHONPATH="${PYTHONPATH}:${HOME_DIR}"
ENV PORT=${PORT}

# Bake the model weights into the image layer (~/.cache/huggingface).
# Deliberately NOT a build cache mount: cache mounts are not committed to image
# layers, so weights downloaded into one would vanish from the final image and
# be re-downloaded over the network at every container start.
RUN python -c "from sentence_transformers import CrossEncoder, SentenceTransformer; \
    SentenceTransformer('${EMBEDDING_MODEL_NAME}', trust_remote_code=True); \
    CrossEncoder('${CROSS_ENCODER_MODEL}')"

WORKDIR ${HOME_DIR}
EXPOSE ${PORT}

CMD ["sh", "-c", "uvicorn main:app --host 0.0.0.0 --port ${PORT}"]
